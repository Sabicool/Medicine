% Duration of action of the intra-operative opioids on a log time axis:
% remifentanil (offset after stopping), alfentanil, fentanyl and morphine
\documentclass[11pt, border=2mm]{standalone}
\usepackage{cicm-figure}
\begin{document}
\begin{tikzpicture}
  % x = 4.0 * log10(minutes)
  \def\s{4.0}
  % Grid and axis
  \foreach \m/\l in {1/1 min, 5/5 min, 15/15 min, 60/1 h, 180/3 h, 300/5 h} {
    \draw[thin, cicmgrey!35] ({\s*log10(\m)},0.2) -- ({\s*log10(\m)},-5.0);
    \node[label, color=cicmgrey, anchor=north] at ({\s*log10(\m)},-5.05) {\l};
  }
  \draw[thin, cicmgrey] (0,-5.0) -- ({\s*log10(300)},-5.0);
  \node[label, anchor=north] at ({\s*log10(17)},-5.55) {duration (log scale)};

  % \bar{y}{from}{to}{name}{use}{range text}
  \newcommand\opioid[6]{%
    \fill[cicmaccent, rounded corners=0.6mm] (0,#1-0.3) rectangle ({\s*log10(#2)},#1+0.3);
    \fill[cicmaccent!35, rounded corners=0.6mm] ({\s*log10(#2)-0.05},#1-0.3) rectangle ({\s*log10(#3)},#1+0.3);
    \node[anchor=east, align=right, font=\small] at (-0.2,#1) {\textbf{#4}\\{\footnotesize\color{cicmgrey}#5}};
    \node[label, anchor=west] at ({\s*log10(#3)+0.1},#1) {#6};}

  \opioid{-0.4}{2}{5}{Remifentanil}{infusion}{minutes after stopping, however long it ran}
  \opioid{-1.6}{10}{15}{Alfentanil}{short procedures}{10--15 min}
  \opioid{-2.8}{30}{45}{Fentanyl}{induction}{30--45 min}
  \opioid{-4.0}{180}{300}{Morphine}{titrated up to 10 mg}{3--5 h}
\end{tikzpicture}
\end{document}

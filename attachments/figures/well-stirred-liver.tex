  \node[label, below] at (4.0,0.68) {$Q_{H}\cdot C_{\text{out}}$};

  % Exit 2: metabolism of unbound drug
  \draw[->, thick, cicmblue] (0.2,-1.3) -- (0.2,-2.7) node[below, label, align=center]
    {Exit 2: metabolism\\$f_{u}\cdot CL_{\text{int}}\cdot C_{\text{out}}$};

  % Interpretation
  \node[label, align=center, anchor=north] at (0,-4.15)

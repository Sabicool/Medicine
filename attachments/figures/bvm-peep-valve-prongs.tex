% Sealed bag-valve mask with a PEEP valve: the valve only caps pressure, so it
% gives PEEP on expiration in a breathing patient, nothing in apnoea, and CPAP
% in apnoea once nasal prongs add a continuous flow under the mask.
% Traces are qualitative; the ~6 cmH2O plateau is from the note.
% Lungs: Servier Medical Art (smart.servier.com), CC BY 4.0 (art/servier-lungs.pdf)
\documentclass[11pt, border=2mm]{standalone}
\usepackage{cicm-figure}
\newcommand\organ[2]{\includegraphics[height=#1]{servier-#2.pdf}}
\begin{document}
\begin{tikzpicture}
  \tikzset{
    tag/.style  = {font=\footnotesize, align=left},
    pipe/.style = {line width=7pt, cicmgrey!30},
  }

  % ---------- Circuit ----------
  \draw[pipe] (1.3,-0.95) -- (1.3,0) -- (8.3,0);
  \draw[pipe] (5.6,0) -- (5.6,-1.5);
  \node[anchor=north] at (1.3,-0.55) {\organ{26mm}{lungs}};

  % Mask
  \draw[thin, fill=cicmpaper] (3.4,-0.55) -- (4.0,-0.25) -- (4.0,0.25) -- (3.4,0.55) -- cycle;
  \node[tag, anchor=south] at (3.7,0.65) {sealed mask};

  % Nasal prongs under the mask
  \draw[->, thick, cicmblue] (3.0,-2.9) -- (3.0,-0.2);
  \node[tag, text=cicmblue, anchor=west] at (3.15,-2.5) {nasal prongs\\ 15 L/min};

  % PEEP valve on the expiratory port
  \draw[thin, draw=cicmaccent, fill=cicmaccenttint, rounded corners=0.6mm] (5.15,-1.5) rectangle (6.05,-2.3);
  \draw[thin, cicmaccent] (5.3,-2.2) -- ++(0.12,0.1) -- ++(-0.12,0.1) -- ++(0.12,0.1) -- ++(-0.12,0.1) -- ++(0.12,0.1);
  \fill[cicmaccent] (5.5,-1.72) rectangle (5.95,-1.62);
  \draw[->, cicmaccent] (5.6,-2.35) -- (5.6,-2.95);
  \node[tag, anchor=west] at (6.2,-1.9) {\textbf{PEEP valve}: spring-loaded,\\ opens only above set pressure};

  % One-way valve and bag
  \draw[thin, fill=white] (7.45,-0.3) rectangle (7.75,0.3);
  \draw (7.6,0.27) -- (7.6,-0.27);
  \node[tag, align=center, anchor=south] at (7.6,0.4) {\textbf{one-way valve}: opens only on\\ inhalation or a bag squeeze};
  \draw[thin, fill=cicmbluetint] (9.6,0) ellipse (1.3 and 0.62);
  \node[tag] at (9.6,0) {bag};
  \draw[->, thin] (11.6,0) -- (10.95,0);
  \node[tag, anchor=west] at (11.65,0) {O$_2$};

  % ---------- Pressure traces ----------
  \pgfplotsset{trace/.style={cicm, scale only axis, width=33mm, height=21mm,
      xmin=0, xmax=8, ymin=-3.5, ymax=12.5, xtick=\empty, ytick={0,10},
      title style={font=\footnotesize, align=center}, xlabel={time},
      xlabel style={font=\footnotesize}, clip=false}}

  \coordinate (g1) at (0.4,-4.25);
  \coordinate (g2) at (4.95,-4.25);
  \coordinate (g3) at (9.5,-4.25);
  \begin{axis}[trace, at={(g1)}, anchor=north west,
      title={Breathing, no prongs}, ylabel={cmH$_2$O}, ylabel style={font=\footnotesize}]
    \addplot[thin, dashed, cicmaccent, domain=0:8] {10};
    \node[tag, text=cicmaccent, anchor=south east] at (axis cs:8,10) {set PEEP};
    \addplot[thick, cicmblue] coordinates {
      (0,10) (0.15,1) (0.35,-1.6) (0.65,-2) (0.95,-1.6) (1.15,-0.3) (1.3,0) (1.45,9) (1.6,10) (4,10)
      (4.15,1) (4.35,-1.6) (4.65,-2) (4.95,-1.6) (5.15,-0.3) (5.3,0) (5.45,9) (5.6,10) (8,10)};
  \end{axis}
  \node[tag, align=center] at (2.05,-7.35) {PEEP on expiration;\\ lost on every inhalation};

  \begin{axis}[trace, at={(g2)}, anchor=north west, title={Apnoea, no prongs}]
    \addplot[thin, dashed, cicmaccent, domain=0:8] {10};
    \addplot[thick, cicmblue, domain=0:8] {0};
  \end{axis}
  \node[tag, align=center] at (6.6,-7.35) {no flow in,\\ so no pressure};

  \begin{axis}[trace, at={(g3)}, anchor=north west, title={Apnoea + prongs}]
    \addplot[thin, dashed, cicmaccent, domain=0:8] {10};
    \addplot[thick, cicmblue, domain=0:8, samples=80] {6*(1-exp(-x/0.9))};
    \node[tag, text=cicmblue, anchor=south] at (axis cs:5,6.3) {CPAP $\approx$ 6};
  \end{axis}
  \node[tag, align=center] at (11.15,-7.35) {prong flow fills the mask\\ until it vents via the valve};
\end{tikzpicture}
\end{document}

\documentclass[11pt, border=2mm]{standalone}
\usepackage{cicm-figure}
\newcommand\organ[2]{\includegraphics[height=#1]{servier-#2.pdf}}
\begin{document}
\begin{tikzpicture}
  % Compartments
  \node[box, minimum width=52mm, minimum height=34mm] (central) at (0,0) {};
  \node[font=\small, anchor=north] at ([yshift=-1mm]central.north) {Central, $V_{1}$};
  \node at (-1.9,-0.05) {\organ{12mm}{heart}};
  \node at (-0.66,-0.05) {\organ{11mm}{brain}};
  \node at (0.62,-0.05) {\organ{10mm}{liver}};
  \node at (1.85,-0.05) {\organ{12mm}{kidney}};
  \node[font=\footnotesize, anchor=south] at ([yshift=1mm]central.south) {plasma and well-perfused organs};

  \node[box, minimum width=52mm, minimum height=34mm, fill=cicmpaper] (peripheral) at (7.4,0) {};
  \node[font=\small, anchor=north] at ([yshift=-1mm]peripheral.north) {Peripheral, $V_{2}$};
  \node at (6.65,-0.05) {\organ{14mm}{muscle}};
  \node at (8.15,-0.05) {\organ{13mm}{fat}};
  \node[font=\footnotesize, anchor=south] at ([yshift=1mm]peripheral.south) {poorly perfused tissue};

  \node[accent box, minimum width=22mm, minimum height=10mm, align=center] (effect) at (0,3.5)
    {Effect site\\[-0.4ex]{\footnotesize $C_{e}$, negligible volume}};

  % Dose
  \draw[->, thick] (-4.6,0) -- (central.west) node[pos=0.4, above, label] {IV dose};

  % Distribution
  \draw[->, thick, cicmblue] ([yshift=3mm]central.east) -- ([yshift=3mm]peripheral.west)
    node[midway, above, label] {$k_{12}$};
  \draw[->, thick, cicmblue] ([yshift=-3mm]peripheral.west) -- ([yshift=-3mm]central.east)
    node[midway, below, label] {$k_{21}$};

  % Effect site equilibration
  \draw[<->, thick, cicmaccent] (central.north) -- (effect.south)
    node[midway, right, label] {$k_{e0}$};

  % Elimination
  \draw[->, thick] (central.south) -- ++(0,-1.4) node[below, label, align=center]
    {Elimination\\$k_{10}=CL/V_{1}$};
\end{tikzpicture}
\end{document}

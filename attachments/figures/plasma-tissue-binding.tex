% Volume of distribution from plasma and tissue binding (Gillette / Øie–Tozer):
% unbound concentrations equalise, so V = V_P + V_T · f_u / f_uT
\documentclass[11pt, border=2mm]{standalone}
\usepackage{cicm-figure}
\begin{document}
\begin{tikzpicture}
  % Plasma
  \node[box, minimum width=38mm, minimum height=30mm, anchor=north] (plasma) at (-3,0) {};
  \node[font=\small, anchor=north] at (plasma.north) {Plasma, $V_{P}$};
  \node[font=\small, align=center] at (-3,-1.55) {Total $C$\\[0.6ex] unbound $f_{u}\cdot C$};

  % Tissue
  \node[box, minimum width=38mm, minimum height=30mm, anchor=north, fill=cicmpaper] (tissue) at (3,0) {};
  \node[font=\small, anchor=north] at (tissue.north) {Tissue, $V_{T}$};
  \node[font=\small, align=center] at (3,-1.55) {Total $C_{T}$\\[0.6ex] unbound $f_{u_{T}}\cdot C_{T}$};

  % Only unbound drug crosses
  \draw[<->, thick, cicmaccent] (plasma.east |- 0,-1.5) -- (tissue.west |- 0,-1.5)
    node[midway, above, label, align=center] {unbound\\drug};

  % Equilibrium
  \node[label, align=center, anchor=north] at (0,-3.35)
    {At equilibrium the unbound concentrations are equal:\quad
     $f_{u}\cdot C=f_{u_{T}}\cdot C_{T}$};
  \node[label, align=center, anchor=north] at (0,-4.15)
    {$V_{D}=\dfrac{\text{Amount}}{C}=\dfrac{C\cdot V_{P}+C_{T}\cdot V_{T}}{C}
     =V_{P}+\dfrac{f_{u}}{f_{u_{T}}}\cdot V_{T}$};
\end{tikzpicture}
\end{document}

% Volume of distribution from plasma and tissue binding (Gillette / Øie–Tozer):

% Artery, muscle and fat: Servier Medical Art (smart.servier.com), CC BY 4.0
% (art/servier-*.pdf)

\begin{document}
\begin{tikzpicture}
  % Panels
  \draw[thin, rounded corners=0.6mm, fill=cicmbluetint] (-4.75,0.65) rectangle (-1.25,-4.1);
  \draw[thin, rounded corners=0.6mm, fill=cicmpaper] (1.25,0.65) rectangle (4.75,-4.1);

  % Plasma
  \node[font=\small] at (-3,0.2) {Plasma, $V_{P}$};
  \node at (-3,-1.35) {\organ{27mm}{artery}};
  \node[font=\small, align=center, anchor=north] at (-3,-2.85)
    {Total $C$\\[0.6ex] unbound $f_{u}\cdot C$};

  % Tissue
  \node[font=\small] at (3,0.2) {Tissue, $V_{T}$};
  \node at (2.25,-1.35) {\organ{25mm}{muscle}};

  \node[font=\small, align=center, anchor=north] at (3,-2.85)
    {Total $C_{T}$\\[0.6ex] unbound $f_{u_{T}}\cdot C_{T}$};

  % Only unbound drug crosses
  \draw[<->, thick, cicmaccent] (-1.2,-1.35) -- (1.2,-1.35)
    node[midway, above, label, align=center] {unbound\\drug};

  % Equilibrium
  \node[label, align=center, anchor=north] at (0,-4.45)
    {At equilibrium the unbound concentrations are equal:\quad

  \node[label, align=center, anchor=north] at (0,-5.25)
    {$V_{D}=\dfrac{\text{Amount}}{C}=\dfrac{C\cdot V_{P}+C_{T}\cdot V_{T}}{C}
     =V_{P}+\dfrac{f_{u}}{f_{u_{T}}}\cdot V_{T}$};
